cmake_minimum_required(VERSION 3.20)
project(OldWestReconnectTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PRIVATE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../OldWest/Source/OldWestGame/Private")
add_library(oldwest_reconnect "${PRIVATE_DIR}/ReconnectStore.cpp")
target_include_directories(oldwest_reconnect PUBLIC "${PRIVATE_DIR}")
if(MSVC)
    target_compile_options(oldwest_reconnect PRIVATE /W4 /WX)
else()
    target_compile_options(oldwest_reconnect PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    target_link_libraries(oldwest_reconnect PUBLIC Advapi32 Crypt32)
    add_executable(oldwest_reconnect_tests reconnect_windows_tests.cpp)
else()
    add_executable(oldwest_reconnect_tests reconnect_tests.cpp)
endif()
target_link_libraries(oldwest_reconnect_tests PRIVATE oldwest_reconnect)
enable_testing()
add_test(NAME oldwest_reconnect COMMAND oldwest_reconnect_tests)
