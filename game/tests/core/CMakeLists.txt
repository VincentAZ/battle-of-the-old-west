cmake_minimum_required(VERSION 3.20)
project(OldWestRulesTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CORE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../OldWest/Source/OldWestGame/Private/Core")
add_library(oldwest_rules "${CORE_DIR}/OldWestRules.cpp" "${CORE_DIR}/OldWestAI.cpp")
target_include_directories(oldwest_rules PUBLIC "${CORE_DIR}")
target_compile_options(oldwest_rules PRIVATE -Wall -Wextra -Wpedantic)
add_executable(oldwest_rules_tests rules_tests.cpp)
target_link_libraries(oldwest_rules_tests PRIVATE oldwest_rules)
enable_testing()
add_test(NAME oldwest_rules COMMAND oldwest_rules_tests "${CMAKE_CURRENT_LIST_DIR}/../../data/cards.json")
add_executable(oldwest_ai_tests ai_tests.cpp)
target_link_libraries(oldwest_ai_tests PRIVATE oldwest_rules)
add_test(NAME oldwest_ai COMMAND oldwest_ai_tests "${CMAKE_CURRENT_LIST_DIR}/../../data/cards.json")

add_executable(oldwest_native_cosmetics_tests ../presentation/native_cosmetics_tests.cpp)
target_link_libraries(oldwest_native_cosmetics_tests PRIVATE oldwest_rules)
add_test(NAME oldwest_native_cosmetics COMMAND oldwest_native_cosmetics_tests "${CMAKE_CURRENT_LIST_DIR}/../../data/cards.json")
